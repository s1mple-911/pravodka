**Yordamchi (yangi, `maqsadHtml` dan yuqorida):**
```js
// Ijrochi chipi. 🔴 Bo'sh bo'lsa ham TEG QAYTARADI (.ijr.unk) — ko'rinishini CSS hal
// qiladi (#journal.ijr-unk). render() ichida shart bo'lmasin: filtr holati = bitta klass.
function ijrHtml(e){
  const raw=String(e.ijrochi||'').trim();
  const sysKey=raw.toLowerCase();
  const isSys = e.ijrochi_tur==='tizim' || Object.prototype.hasOwnProperty.call(IJR_SYS,sysKey)
             || String(e.source||'')==='aros_auto';
  if(!raw && !isSys)
    return `<span class="ijr unk" title="Ijrochi qayd etilmagan"><i data-lucide="user"></i><span class="ijr-nm">Noma'lum</span></span>`;
  if(isSys){
    const lbl=IJR_SYS[sysKey]||raw||'Tizim';
    return `<span class="ijr sys" title="Avtomatik yozuv · ${esc(raw||e.source||'tizim')}"><i data-lucide="bot"></i><span class="ijr-nm">${esc(lbl)}</span></span>`;
  }
  return `<span class="ijr" title="Yozuvni kiritdi: ${esc(raw)}"><i data-lucide="user"></i><span class="ijr-nm">${esc(raw)}</span></span>`;
}
```

**Uchala holat — tayyor HTML (dizayn tekshiruvi uchun):**

```html
<span class="ijr sys" title="Avtomatik yozuv · aros_sync">
```

### 2.5 🔴 7-filtr sig'adimi? — ANIQ javob

## 4. Excel — jurnal eksportiga `Ijrochi` ustuni

🔴 **Ikkala massiv BITTA tahrirda o'zgaradi**, aks holda ustunlar siljiydi:

**`xlsRow()` (2351):** `e.source||''` dan **oldin**:

```js
// Excel uchun sof matn (chip emas). Noma'lum = BO'SH katak — "Noma'lum" so'zi
// filtrni/pivotni ifloslantiradi; bo'sh katak Excel'da to'g'ri "(blank)" bo'ladi.
function ijrText(e){
  const raw=String(e.ijrochi||'').trim();
  if(/^[0-9a-fA-F-]{36}$/.test(raw)) return '';
  const k=raw.toLowerCase();
  if(Object.prototype.hasOwnProperty.call(IJR_SYS,k)) return IJR_SYS[k];
  if(!raw && String(e.source||'')==='aros_auto') return IJR_SYS.aros_sync;
  return raw;
}
```
